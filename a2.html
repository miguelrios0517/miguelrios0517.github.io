//define pin number for components 
int pinButton = 13; //button
int red = 11; //red LED
int green = 10; //green LED
int blue = 9; //blue LED

int buttonState = 0; //initialize button state (will store HIGH or LOW)

// initialize random variables
int R; // value for red LED
int G; // value for green LED
int B; // value for blue LED

void setup() {
  // put your setup code here, to run once:
  pinMode(pinButton, INPUT); //setup button pin as INPUT (check if button pressed, if there is current)
  pinMode(red, OUTPUT); //setup red LED pin as OUTPUT 
  pinMode(green, OUTPUT); //set green LED pin as OUTPUT
  pinMode(blue, OUTPUT); //set blue LED pin as OUTPUT 
}

void loop() {
  // put your main code here, to run repeatedly:
  buttonState = digitalRead(pinButton); //read if button pin receives current (i.e button is pressed)
  //Serial.println();
  if(buttonState == HIGH) { // if button is pressed
    for (int i = 0; i<= 255; i++) { // iterate from 0 to 255, increase brightness
      analogWrite(red,int(i/(255/R))); // sets duty cycle of red LED -> current loop * (value of random color/255)
      analogWrite(green, int(i/(255/G))); // sets duty cycle of green LED
      analogWrite(blue, int(i/(255/B))); // sets duty cycle of blue LED
    }
    for (int i = 255; i>= 0; i--) { // iterate from 255 to 0, decrease brightness
      analogWrite(red,int(i/(255/R))); // sets duty cycle of red LED
      analogWrite(green, int(i/(255/G))); // sets duty cycle of green LED
      analogWrite(blue, int(i/(255/B))); // sets duty cycle of blue LED
    }
  } else { // else if button not pressed
    analogWrite(red,0); // set duty cycle of red LED to 0
    analogWrite(green,0); // set duty cycle of green LED to 0
    analogWrite(blue,0); // set duty cycle of blue LED to 0
    randomSeed(analogRead(0));
    R = random(255); // sets random red value
    G = random(255); // sets random green value
    B = random(255); // sets random blue value
  }
}

Question 2: Given your schematic, circuit, and firmware, assuming the only thing that draws current is your LEDs, how long would your circuit run if powered by a 1200 mAh battery?


Theoretically, if all 3 LEDS are on, the current for the circuit equals 20mA + 20mA + 20mA = 60mA.
1200 maH = 60mA * hours
20 = hours
The circuit would run for 20 hours.
